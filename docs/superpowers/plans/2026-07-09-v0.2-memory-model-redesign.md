# v0.2 记忆模型重构实现计划

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐任务实现此计划。步骤使用复选框（`- [ ]`）语法来跟踪进度。

**目标：** 重构冲突/伏笔/钩子/弧光四实体职能边界，新增钩子和角色弧光两张表，改造 create-flow 让各 Agent 按边界读写实体，解决 v0.1 中三类实体职能重叠问题。

**技术栈：** TypeScript, better-sqlite3, Fastify, React, TanStack Query

```typescript
// ── 钩子（v0.2：章末悬念驱动实体）──

/** 钩子状态：thrown（抛出）→ fulfilled（兑现）/ abandoned（弃用） */
export type HookStatus = 'thrown' | 'fulfilled' | 'abandoned'

/**
 * 章尾钩子实体（v0.2 新增）
 *
 * 职能边界：节奏层，追踪章末悬念的「抛出→兑现」闭环。
 * 与伏笔/冲突正交：伏笔是信息层（埋设-回收），冲突是张力层（有对抗方），
 * 钩子是节奏层（单章末的读者驱动设计）。
 *
 * 创建时机：Director 规划本章时创建（status=thrown）
 * 兑现时机：Memory Agent 检测到后续章节承接时标记 fulfilled
 * 关联建立：Memory Agent 创建伏笔/冲突后回填 linked_*_id
 */
export interface NovelHook {
  id: string
  projectId: string
  /** 抛出钩子的章节号 */
  chapterNumber: number
  hookType: ChapterEndingHookType
  /** 钩子内容描述（Director 规划时填） */
  description: string
  status: HookStatus
  /** 兑现章节号（fulfilled 时填） */
  fulfilledAtChapter: number | null
  /** 兑现方式描述（Writer 兑现后填） */
  fulfilledDescription: string | null
  /** 可选关联伏笔（钩子是伏笔的埋设） */
  linkedForeshadowId: string | null
  /** 可选关联冲突（钩子是冲突升级） */
  linkedConflictId: string | null
  createdAt: number
  updatedAt: number
}

// ── 角色弧光（v0.2：角色内心阶段演进）──

/** 弧光类型：growth成长 / fall堕落 / redemption救赎 / shift转变 */
export type ArcType = 'growth' | 'fall' | 'redemption' | 'shift'

/** 弧光阶段历史记录项 */
export interface ArcStageHistoryEntry {
  stage: string
  triggerEvent: string
  description: string
  chapterNumber: number
  changedAt: number
}

/**
 * 角色弧光实体（v0.2 新增）
 *
 * 职能边界：成长层，追踪角色内心的阶段演进。
 * 与 character_history 正交：character_history 记录横向状态变更（currentState/goal），
 * 弧光记录纵向成长线（只在阶段转换时更新）。
 *
 * 1:1 关联角色（一个角色一条主弧光线，多弧光留到 v0.3）。
 * character.arcStage 字段保留作弧光当前阶段的镜像。
 */
export interface NovelCharacterArc {
  id: string
  projectId: string
  characterId: string
  arcType: ArcType
  currentStage: string
  stageHistory: ArcStageHistoryEntry[]
  description: string | null
  createdAt: number
  updatedAt: number
}

/** 弧光阶段模型（按 arcType 对应不同阶段序列） */
export const ARC_STAGE_MODELS: Record<ArcType, string[]> = {
  growth: ['denial', 'resistance', 'acceptance', 'transformation'],
  fall: ['temptation', 'rationalization', 'corruption', 'point_of_no_return'],
  redemption: ['guilt', 'atonement', 'sacrifice', 'redemption'],
  shift: ['old_framework', 'disruption', 'exploration', 'new_framework'],
}
```

```typescript
  // 钩子 CRUD（v0.2）
  // ═══════════════════════════════════════════════════════════════

  private rowToHook(row: HookRow): NovelHook {
    return {
      id: row.id,
      projectId: row.project_id,
      chapterNumber: row.chapter_number,
      hookType: row.hook_type as NovelHook['hookType'],
      description: row.description,
      status: row.status as NovelHook['status'],

      fulfilledDescription: row.fulfilled_description,
      linkedForeshadowId: row.linked_foreshadow_id,
      linkedConflictId: row.linked_conflict_id,
      createdAt: row.created_at,
      updatedAt: row.updated_at,
    }
  }

  listHooks(projectId: string): NovelHook[] {
    const rows = this.db.prepare('SELECT * FROM novel_hooks WHERE project_id = ? ORDER BY chapter_number ASC, created_at ASC').all(projectId) as HookRow[]
    return rows.map(r => this.rowToHook(r))
  }

  /** 查询指定章节抛出的钩子 */
  listHooksByChapter(projectId: string, chapterNumber: number): NovelHook[] {
    const rows = this.db.prepare('SELECT * FROM novel_hooks WHERE project_id = ? AND chapter_number = ? ORDER BY created_at ASC').all(projectId, chapterNumber) as HookRow[]
    return rows.map(r => this.rowToHook(r))
  }

  /** 查询指定章节之前抛出且未兑现的钩子（供 Writer 承接） */
  listUnfulfilledHooksBefore(projectId: string, beforeChapter: number): NovelHook[] {
    const rows = this.db.prepare(
      'SELECT * FROM novel_hooks WHERE project_id = ? AND chapter_number < ? AND status = ? ORDER BY chapter_number DESC, created_at ASC'
    ).all(projectId, beforeChapter, 'thrown') as HookRow[]
    return rows.map(r => this.rowToHook(r))
  }

  getHook(id: string): NovelHook | undefined {
    const row = this.db.prepare('SELECT * FROM novel_hooks WHERE id = ?').get(id) as HookRow | undefined
    return row ? this.rowToHook(row) : undefined
  }

  createHook(hook: Omit<NovelHook, 'createdAt' | 'updatedAt'>): NovelHook {
    const ts = now()
    const full: NovelHook = { ...hook, createdAt: ts, updatedAt: ts }
    this.db.prepare(`
      INSERT INTO novel_hooks (id, project_id, chapter_number, hook_type, description, status,
      fulfilled_at_chapter, fulfilled_description, linked_foreshadow_id, linked_conflict_id,
      created_at, updated_at)
      VALUES (@id, @project_id, @chapter_number, @hook_type, @description, @status,
      @fulfilled_at_chapter, @fulfilled_description, @linked_foreshadow_id, @linked_conflict_id,
      @created_at, @updated_at)
    `).run({
      id: full.id, project_id: full.projectId, chapter_number: full.chapterNumber,
      hook_type: full.hookType, description: full.description, status: full.status,
      fulfilled_at_chapter: full.fulfilledAtChapter,
      fulfilled_description: full.fulfilledDescription,
      linked_foreshadow_id: full.linkedForeshadowId,
      linked_conflict_id: full.linkedConflictId,
      created_at: full.createdAt, updated_at: full.updatedAt,
    })
    return full
  }

  updateHook(id: string, patch: Partial<NovelHook>): NovelHook | undefined {
    const existing = this.db.prepare('SELECT * FROM novel_hooks WHERE id = ?').get(id) as HookRow | undefined
    if (!existing) return undefined
    const current = this.rowToHook(existing)
    const updated = { ...current, ...patch, id, updatedAt: now() }
    this.db.prepare(`
      UPDATE novel_hooks SET hook_type=@hook_type, description=@description, status=@status,
      fulfilled_at_chapter=@fulfilled_at_chapter, fulfilled_description=@fulfilled_description,
      linked_foreshadow_id=@linked_foreshadow_id, linked_conflict_id=@linked_conflict_id,
      updated_at=@updated_at WHERE id=@id
    `).run({
      id, hook_type: updated.hookType, description: updated.description, status: updated.status,
      fulfilled_at_chapter: updated.fulfilledAtChapter,
      fulfilled_description: updated.fulfilledDescription,
      linked_foreshadow_id: updated.linkedForeshadowId,
      linked_conflict_id: updated.linkedConflictId,
      updated_at: updated.updatedAt,
    })
    return updated
  }

  deleteHook(id: string): boolean {
    const result = this.db.prepare('DELETE FROM novel_hooks WHERE id = ?').run(id)
    return result.changes > 0
  }
```

```typescript
function HookTrackRow({
  hooks, chapterByNumber, visibleChapterColumns, visibleRange, maxChapter, cellWidth, playheadChapter,
}: HookTrackRowProps & { hooks: NovelHook[] }): ReactElement | null {
  const hooksByChapter = new Map<number, NovelHook[]>()
  for (const h of hooks) {
    const arr = hooksByChapter.get(h.chapterNumber) ?? []
    arr.push(h)
    hooksByChapter.set(h.chapterNumber, arr)

  // 仅当存在任何钩子时显示该轨道行
  if (hooks.length === 0) return null

      <div className="sticky left-0 z-30 bg-surface-1 flex items-center justify-end pr-2 border-r border-border shrink-0" style={{ width: `${SIDEBAR_W}px` }}>
```

## 任务 9：数据迁移 CLI 工具

**文件：**
- 创建：`apps/agent-server/scripts/migrate-v0.2-memory-model.mjs`

- [ ] **步骤 1：创建迁移脚本**

```javascript
#!/usr/bin/env node
/**
 * v0.2 数据迁移 CLI：将 v0.1 错位数据归位到正确的实体表
 *
 * 迁移内容：
 * 1. type=inner 且无对抗方的冲突（3条）→ 角色弧光
 * 2. 散落到 conflicts 的谜题 → foreshadowing（半自动，需确认）
 * 3. 补填 chapter_ending_hook 镜像
 *
 * 用法：node scripts/migrate-v0.2-memory-model.mjs <projectId> [--dry-run]
 */
import Database from 'better-sqlite3'
import { randomUUID } from 'node:crypto'
import { join } from 'node:path'
import { homedir } from 'node:os'

const projectId = process.argv[2]
const dryRun = process.argv.includes('--dry-run')

if (!projectId) {
  console.error('用法：node migrate-v0.2-memory-model.mjs <projectId> [--dry-run]')
  process.exit(1)
}

const dbPath = join(homedir(), '.pi', 'wenlang-agent.db')
const db = new Database(dbPath, dryRun ? { readonly: true } : {})

console.log(`项目 ID：${projectId}`)
console.log(`模式：${dryRun ? 'DRY RUN（只读，不写入）' : '执行（写入）'}`)
console.log('---')

// 迁移1：type=inner 且 participants 为空的冲突 → 角色弧光
const innerConflicts = db.prepare(
  "SELECT * FROM novel_conflicts WHERE project_id = ? AND type = 'inner'"
).all(projectId)

console.log(`\n[迁移1] 发现 ${innerConflicts.length} 条 type=inner 的冲突`)

for (const c of innerConflicts) {
  let participants = []
  try { participants = JSON.parse(c.participants) } catch {}
  const hasParticipants = Array.isArray(participants) && participants.length > 0

  console.log(`  - ${c.id}: ${c.description.slice(0, 60)}`)
  console.log(`    participants: ${JSON.stringify(participants)}`)

  if (!hasParticipants) {
    // 无对抗方 → 转角色弧光
    // 从 participants 尝试提取 characterId，若无则跳过（需人工处理）
    const characterId = typeof participants[0] === 'string' ? participants[0] : null
    if (!characterId) {
      console.log(`    ⚠️ 无法自动提取 characterId，跳过（需人工处理）`)
      continue
    }

    // 推断 arcType（简单启发式，需人工确认）
    const arcType = c.description.includes('认知') || c.description.includes('信念') ? 'shift' : 'growth'
    const stage = arcType === 'shift' ? 'disruption' : 'resistance'

    console.log(`    → 转为角色弧光：characterId=${characterId}, arcType=${arcType}, stage=${stage}`)

    if (!dryRun) {
      const arcId = randomUUID()
      const chapterSpans = JSON.parse(c.chapter_spans || '[]')
      const chapterNumber = chapterSpans[0]?.[0] ?? 1
      db.prepare(`
        INSERT INTO novel_character_arcs (id, project_id, character_id, arc_type, current_stage, stage_history, description, created_at, updated_at)
        VALUES (?, ?, ?, ?, ?, ?, ?, unixepoch(), unixepoch())
      `).run(arcId, projectId, characterId, arcType, stage,
        JSON.stringify([{ stage, triggerEvent: c.description, description: c.description, chapterNumber, changedAt: Date.now() }]),
        c.description)
      console.log(`    ✓ 已创建弧光 ${arcId}`)
    }
  } else {
    console.log(`    → 有对抗方，保留为冲突`)
  }
}

// 迁移2：补填 chapter_ending_hook 镜像
const hooksByChapter = {}
const hooks = db.prepare('SELECT * FROM novel_hooks WHERE project_id = ?').all(projectId)
for (const h of hooks) {
  if (!hooksByChapter[h.chapter_number]) hooksByChapter[h.chapter_number] = h
}
console.log(`\n[迁移2] 发现 ${hooks.length} 条钩子实体，补填 chapter_ending_hook 镜像`)

for (const [chNum, hook] of Object.entries(hooksByChapter)) {
  console.log(`  - 第${chNum}章：${hook.hook_type}`)
  if (!dryRun) {
    db.prepare('UPDATE novel_chapters SET chapter_ending_hook = ? WHERE project_id = ? AND chapter_number = ?')
      .run(hook.hook_type, projectId, chNum)
  }
}

console.log(`\n${dryRun ? 'DRY RUN 完成，未写入任何数据' : '迁移完成'}`)
db.close()
```
