> **面向 AI 代理的工作者：** 使用 superpowers:executing-plans 或 superpowers:subagent-driven-development 逐任务实现此计划。步骤使用复选框（`- [ ]`）跟踪进度。

**技术栈：** TypeScript、better-sqlite3、Fastify、React + TanStack Query、Vitest。

// apps/agent-server/src/core/observability/model-call-telemetry.ts
import { createHash } from 'node:crypto'
import {
  supportsAnthropicCacheControl,
  usesOpenAiPromptCacheKey,
  type CacheControlTarget,
} from '../plugin/cache-control-support.js'

export type CacheProtocol =
  | 'anthropic-cache-control'
  | 'openai-prompt-cache-key'
  | 'automatic-prefix'
  | 'none'
  | 'unknown'

export type TelemetryDataQuality = 'complete' | 'partial' | 'no_data'

/** 判定目标使用的缓存协议（口径复用 cache-control-support，避免分叉） */
export function resolveCacheProtocol(target: CacheControlTarget): CacheProtocol {
  const hasInfo = target.provider || target.baseUrl || target.modelId
  if (!hasInfo) return 'unknown'
  if (supportsAnthropicCacheControl(target)) return 'anthropic-cache-control'
  if (usesOpenAiPromptCacheKey(target)) return 'openai-prompt-cache-key'
  return 'automatic-prefix'
}

export interface RequestLayers {
  system: string
  shared: string
  identity: string
  dynamic: string
}

/** 请求指纹：分层规范化文本的 sha256 前 16 位。只存摘要不存原文 */
export function buildRequestFingerprint(layers: RequestLayers): string {
  const h = createHash('sha256')
  h.update(`system:${layers.system.length}:`)
  h.update(layers.system)
  h.update(`shared:${layers.shared.length}:`)
  h.update(layers.shared)
  h.update(`identity:${layers.identity.length}:`)
  h.update(layers.identity)
  h.update(`dynamic:${layers.dynamic.length}:`)
  h.update(layers.dynamic)
  return h.digest('hex').slice(0, 16)
}

/** 实际缓存读取率；输入数据缺失返回 null */
export function cacheHitRate(cacheReadTokens: number | null, inputTokens: number | null): number | null {
  if (inputTokens == null || cacheReadTokens == null) return null
  const prompt = cacheReadTokens + inputTokens
  if (prompt <= 0) return null
  return cacheReadTokens / prompt
}

/** 缓存前缀覆盖率 = 实测 cacheRead / 期望前缀 token */
export function prefixCoverage(cacheReadTokens: number | null, expectedTokens: number | null): number | null {
  if (expectedTokens == null || expectedTokens <= 0 || cacheReadTokens == null) return null
  return cacheReadTokens / expectedTokens
}

export interface UsageFields {
  inputTokens: number | null
  outputTokens: number | null
  cacheReadTokens: number | null
  cacheWriteTokens: number | null
}

/** 数据质量：四项齐备 complete；有 input 但其余缺 partial；input 也没有 no_data */
export function classifyDataQuality(u: UsageFields): TelemetryDataQuality {
  if (u.inputTokens == null || u.inputTokens === 0 && u.cacheReadTokens == null) return 'no_data'
  if (u.outputTokens == null && u.cacheReadTokens == null && u.cacheWriteTokens == null) return 'partial'
  return 'complete'
}
```

// ── 观测 API（缓存与监控一体化）──

export interface ObservabilityFilterParams {
  from?: number
  to?: number
  chapterNumber?: number
  agentType?: string
  limit?: number
}

export interface ObservabilityOverview {
  totalCalls: number
  successRate: number
  errorRate: number
  timeoutRate: number
  inputTokens: number
  outputTokens: number
  cacheReadTokens: number
  cacheWriteTokens: number
  cacheHitRate: number | null
  avgDurationMs: number
  totalCostUsd: number
  dataQuality: { completeCalls: number; partialCalls: number; noDataCalls: number }
  timestamp: number
}

export interface ObservabilityCache {
  overallCoverage: number | null
  verdict: 'healthy' | 'partial' | 'miss' | 'no_data'
  potentialSavedTokens: number
  notes: string[]
  chapters: Array<{ chapterNumber: number; verdict: string; followUpCoverage: number | null }>
  byAgent: Array<{ agentType: string; hitRate: number | null; calls: number }>
  protocolDistribution: Record<string, number>
  fingerprintStability: { totalCalls: number; distinctFingerprints: number }
  timestamp: number
}

export interface ObservabilityAgentRow {
  agentType: string
  calls: number
  successRate: number
  inputTokens: number
  outputTokens: number
  cacheHitRate: number | null
  avgDurationMs: number
  costUsd: number
  lastError: string | null
}

export interface ObservabilityTimelineRow {
  id: string
  chapterNumber: number
  agentType: string
  phase: string
  status: string
  durationMs: number
  inputTokens: number
  cacheReadTokens: number
  cacheProtocol: string | null
  cacheHitRate: number | null
  createdAt: number
}

function obsQuery(params?: ObservabilityFilterParams): string {
  const s = new URLSearchParams()
  if (params?.from != null) s.set('from', String(params.from))
  if (params?.to != null) s.set('to', String(params.to))
  if (params?.chapterNumber != null) s.set('chapterNumber', String(params.chapterNumber))
  if (params?.agentType) s.set('agentType', params.agentType)
  if (params?.limit != null) s.set('limit', String(params.limit))
  const q = s.toString()
  return q ? `?${q}` : ''
}

export function getObservabilityOverview(projectId: string, params?: ObservabilityFilterParams) {
  return getApi<ObservabilityOverview>(`${BASE}/projects/${projectId}/observability/overview${obsQuery(params)}`)
}
export function getObservabilityCache(projectId: string, params?: ObservabilityFilterParams) {
  return getApi<ObservabilityCache>(`${BASE}/projects/${projectId}/observability/cache${obsQuery(params)}`)
}
export function getObservabilityAgents(projectId: string, params?: ObservabilityFilterParams) {
  return getApi<ObservabilityAgentRow[]>(`${BASE}/projects/${projectId}/observability/agents${obsQuery(params)}`)
}
export function getObservabilityTimeline(projectId: string, params?: ObservabilityFilterParams) {
  return getApi<ObservabilityTimelineRow[]>(`${BASE}/projects/${projectId}/observability/timeline${obsQuery(params)}`)
}

// observability-timeline.tsx
import { useQuery } from '@tanstack/react-query'
import type { ReactElement } from 'react'
import { useNovelDataStore } from '@/stores/novel-data-store'
import { getObservabilityTimeline } from '@/api/novel-client'

const AGENT_COLORS: Record<string, string> = {
  director: 'bg-blue-500', writer: 'bg-emerald-500',
  reviewer: 'bg-orange-500', reviser: 'bg-purple-500', memory: 'bg-gray-500',
}

/** 按时间的调用趋势（纯 CSS 条形，不引入图表库） */
export function ObservabilityTimeline(): ReactElement {
  const projectId = useNovelDataStore((s) => s.currentProjectId)
  const q = useQuery({
    queryKey: ['novel-observability-timeline', projectId],
    queryFn: () => getObservabilityTimeline(projectId!, { limit: 50 }),
    enabled: !!projectId,
    refetchInterval: 60_000,
    placeholderData: (prev) => prev,
  })
  if (!projectId) return <p className="p-3 text-xs text-muted-foreground">请先选择项目</p>
  if (!q.data) return <p className="p-3 text-xs text-muted-foreground">{q.isLoading ? '加载中...' : '暂无数据'}</p>
  const max = Math.max(1, ...q.data.map((r) => r.inputTokens + r.cacheReadTokens))
  return (
    <div className="space-y-1 p-3">
      <p className="text-xs font-medium text-text">最近调用趋势</p>
      {q.data.map((r) => (
        <div key={r.id} className="flex items-center gap-2 text-nano">
          <span className="w-14 shrink-0 text-muted-foreground tabular-nums">第{r.chapterNumber}章</span>
          <span className="w-10 shrink-0">{r.agentType}</span>
          <div className="h-1.5 flex-1 overflow-hidden rounded-full bg-muted">
            <div
              className={cn('h-full', r.status === 'success' ? (AGENT_COLORS[r.agentType] ?? 'bg-primary') : 'bg-destructive')}
              style={{ width: `${((r.inputTokens + r.cacheReadTokens) / max) * 100}%` }}
            />
          </div>
          <span className="w-14 shrink-0 text-right text-muted-foreground tabular-nums">
            {r.cacheHitRate != null ? `${(r.cacheHitRate * 100).toFixed(0)}%` : '—'}
          </span>
        </div>
      ))}
    </div>
  )
}
```
